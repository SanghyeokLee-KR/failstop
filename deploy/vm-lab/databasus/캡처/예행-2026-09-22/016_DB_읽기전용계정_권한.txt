echo "  PostgreSQL 16 부터 상속 여부는 역할 속성이 아니라 이 GRANT 기록에 고정된다."

echo
echo "=== 권한별 테이블 수 (명시 GRANT 기준) ==="
sudo -u postgres psql -d exchange -c "
SELECT count(*) AS 전체테이블,
  count(*) FILTER (WHERE has_table_privilege('databasus_ro', format('%I.%I', table_schema, table_name), 'INSERT')) AS 삽입가능,
  count(*) FILTER (WHERE has_table_privilege('databasus_ro', format('%I.%I', table_schema, table_name), 'UPDATE')) AS 수정가능,
  count(*) FILTER (WHERE has_table_privilege('databasus_ro', format('%I.%I', table_schema, table_name), 'DELETE')) AS 삭제가능,
  count(*) FILTER (WHERE has_table_privilege('databasus_ro', format('%I.%I', table_schema, table_name), 'TRUNCATE')) AS 비우기가능
FROM information_schema.tables WHERE table_schema='public' AND table_type='BASE TABLE';"
echo
echo "=== 실제 동작 확인: 역할을 갈아입고 해 본다 ==="
echo "--- 읽기 (되어야 한다)"
sudo -u postgres psql -d exchange -c "SET ROLE databasus_ro; SELECT count(*) AS 원장기록 FROM ledger_entries;" 2>&1 | tail -4
sudo -u postgres psql -d exchange -c "SET ROLE databasus_ro; SELECT count(*) AS 잔고행 FROM balances;" 2>&1 | tail -4
echo "--- 쓰기 (거부되어야 한다)"
sudo -u postgres psql -d exchange -c "SET ROLE databasus_ro; DELETE FROM ledger_entries;" 2>&1 | tail -2
sudo -u postgres psql -d exchange -c "SET ROLE databasus_ro; UPDATE balances SET available = 0;" 2>&1 | tail -2
echo "--- 객체 생성 (거부되어야 한다)"

echo "--- 슈퍼유저 여부"
