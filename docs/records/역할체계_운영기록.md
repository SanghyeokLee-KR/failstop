  독립성은 인격이 아니라 권한과 지시문에서만 나온다. 공통 오류 모드는 공유될 수 있다.

foreach ($k in $map.Keys) {
  if ($map[$k].Count -eq 1) { $t = @($map[$k])[0]; $single[$t] = 1 + $single[$t]; $singleTotal++ }
  elseif ($map[$k].Count -eq 0) { $empty++ } else { $multi++ }
}
"$($map.Count) $singleTotal $multi $empty"                    # 314 181 131 2
$single.GetEnumerator() | Sort-Object @{E='Value';Descending=$true}, Name
# web 79, server 32, docs 31, harness 30, .gitignore 3, 제출_20260901 3, deploy 2, scripts 1
